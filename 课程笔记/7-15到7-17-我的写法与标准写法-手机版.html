7-15 · 数列求和难点：看清 An 本身已经是一层求和

题目规则

输入一个正整数 n（1 ≤ n ≤ 3000）。数列第 n 项的定义是：

An = 1×2×3 + 2×3×4 + 3×4×5 + … + (n−2)×(n−1)×n     (n ≥ 3)
An = 0                                                (n = 1, 2)

要求输出这个数列的前 n 项和：S = A1 + A2 + … + An。题目提示结果很大，建议用 long long。

这题真正的坑在第一步读题：An 本身已经是一串乘积的和。它不是"第 n 项 = (n−2)(n−1)n"，而是"第 n 项 = 从 1×2×3 一路加到 (n−2)(n−1)n"。

所以整道题是两层求和：内层算单项，外层算前 n 项和。漏看那个加号、写成一层循环，样例能过但大 n 全错。这一处你理解对了——你的双层循环正好对应原定义。

你的写法

#include <stdio.h>
int main(void)
{
    long long int j=0;
    int n=0;
    int i=0;
    long long int a=0;
    long long int S=0;

    scanf("%d",&n);
    if(n<=2) printf("0");
    else
    {
    for(i=1;i<=n;i++)
    {
        for(j=1;j<=i-2;j++)
        {
           a=a+j*(j+1)*(j+2);
        }
        S=S+a;
        a=0;
    }
            printf("%lld\n",S);
    }

    return 0;
}

先说一句：a=0; 放在 S=S+a; 之后、下一轮之前，这个位置是对的——每项独立累加，互不污染。这点你没写错。

标准写法 · 4.0（递推）

#include<stdio.h>
int main(void)
{
    int n, i;
    long long Ai = 0;
    long long sum = 0;

    scanf("%d", &n);
    if (n <= 2) {
        printf("0\n");
        return 0;
    }

    for (i = 3; i <= n; i++) {
        Ai += (long long)(i - 2) * (i - 1) * i;
        sum += Ai;
    }

    printf("%lld\n", sum);
    return 0;
}

关键在这一行：Ai += (long long)(i-2) * (i-1) * i;

你的写法是「每算一项，都从头把 i−2 个乘积重新加一遍」；标准写法是「第 i 项 = 第 i−1 项 + 一个新的乘积」。所以它只需要加一个新数，不用重算。这一下就把 O(n²) 变成 O(n)。

标准写法 2.0（和你的几乎一样，只是收拾干净了） 
    for (i = 3; i <= n; i++) {
        a = 0;
        for (j = 1; j <= i - 2; j++) {
            a += j * (j + 1) * (j + 2);
        }
        S += a;
    }

外层从 3 开始（跳过必然为 0 的前两项），其余和你的完全同构。3.0 只是把 Ai = 0; 挪到外层循环开头，逻辑一致。

差异清单

① 输出少了一个换行

你的printf("0");
标准printf("0\n");

n≤2 这条支路只有一种输出，很容易被忽略。有些判题系统对行尾换行不敏感，有些会判错——不赌，直接对齐。

② 外层从 1 开始，前两轮空转

你的for(i=1; i<=n; i++)
标准for(i=3; i<=n; i++)

i=1、i=2 时内层条件 j<=i-2 直接不成立，白跑两圈。无害，但说明"循环从哪开始"这件事还没被当成设计决策。

③ 真正有分量的差距：复杂度

你的O(n²) —— 每个 i 重算一遍
标准 4.0O(n) —— 只加一个新项

这不是编码能力的差距，是数学眼力的差距。看出"第 i 项 = 第 i−1 项 + 一个新乘积"，需要的是把它当递推数列看。这类提升只能靠多见，急不来——但你可以每道题多问一句：这一项能不能从上一项推出来？

④ 一个不算错的细节

你的long long j（只做计数）
标准int j

j 最大只到 n，用 long long 浪费但无害。真正需要 64 位的是累加变量 a 和 S——这两个你都写对了。

7-16 · 倍数统计难点：两种"判定重叠"的流派

规则（反推）

读入一串正整数，读到 0 结束（0 不计入）。统计每个数是 3、5、7 中恰好一个的倍数，分别记三类，最后输出三类各占全部数字的百分比，保留两位小数并带 %。

你的写法

#include<stdio.h>
int main(void)
{
    int cnt;
    int a;
    double A=0.0;
    double B=0.0;
    double C=0.0;
    cnt=0;
    double x;
    double y;
    double z;
    for(scanf("%d",&a);a!=0;scanf("%d",&a))
        {
            cnt++;
            if(a%15==0||a%21==0||a%35==0)
                continue;
            else if(a%3==0) A++;
                if(a%5==0) B++;
                if(a%7==0) C++;
        }
    x=(A/cnt)*100.0;
    y=(B/cnt)*100.0;
    z=(C/cnt)*100.0;
    printf("%.2lf%%\n%.2lf%%\n%.2lf%%\n",x,y,z);

    return 0;
}

标准写法 · 2.0（计数派）

#include <stdio.h>
int main(void)
{
    int a;
    int total = 0;
    int cnt3 = 0, cnt5 = 0, cnt7 = 0;

    while (scanf("%d", &a) == 1 && a != 0) {
        total++;

        int match = 0;
        if (a % 3 == 0) match++;
        if (a % 5 == 0) match++;
        if (a % 7 == 0) match++;

        if (match == 1) {
            if (a % 3 == 0) cnt3++;
            if (a % 5 == 0) cnt5++;
            if (a % 7 == 0) cnt7++;
        }
    }

    printf("%.2lf%%\n%.2lf%%\n%.2lf%%\n",
           (double)cnt3 / total * 100.0,
           (double)cnt5 / total * 100.0,
           (double)cnt7 / total * 100.0);
    return 0;
}
标准写法 · 4.0（枚举派，和你同一思路） 
    while (scanf("%d", &a) == 1 && a != 0) {
        total++;
        if (a % 15 == 0 || a % 21 == 0 || a % 35 == 0) {
            continue;
        }
        if (a % 3 == 0) cnt3++;
        if (a % 5 == 0) cnt5++;
        if (a % 7 == 0) cnt7++;
    }

和你的判法完全一样，唯一多的是 scanf(...) == 1。

差异清单

① 最要紧的一条：输入没有保护

你的for(scanf(...); a!=0; scanf(...))
三份标准while (scanf(...) == 1 && a != 0)

输入读到末尾时，scanf 返回 0 或 -1，a 保持上一次的值不变。只要那个值不是 0，a != 0 永远成立，循环就停不下来。PTA 因为题目保证以 0 结尾，碰不到；但这是"靠题目保证才成立"，不是"靠代码保证"。三份标准答案都做了这件事，只有你没做。

② 四份代码都没处理的一种输入

输入只有 0total = 0 → 除以零
结果输出 nan%

一个有效数字都没有的时候，cnt3 / total 就是除以零，打出来是 nan%。这是标准答案也漏掉的一格——抄标准答案学不到它，只能靠自己的情况表补。

③ 两种流派：枚举 vs 计数

枚举派（你 / 4.0）%15 || %21 || %35
计数派（2.0 / 3.0）match == 1

枚举派靠"列出所有两两组合"，计数派靠"数有几个条件成立"。三个条件时两者都对，但条件加到四个（2、3、5、7），枚举派要写 11 项排除条件，漏一项就错；计数派只加一行 if。可扩展性差着一个量级。

④ 计数器用了 double

你的double A, B, C
标准int cnt3, cnt5, cnt7

结果是对的，但计数是整数操作，用 double 会让读的人误以为这里需要小数精度。类型应该表达意图——只在最后算百分比时转成 double 就够。

⑤ 缩进把逻辑说错了

你的else if(...) A++;
　　if(...) B++;
　　if(...) C++;
实际语义三个独立的 if

后面两个 if 缩进比 else if 还深，看起来像是它的子句，其实是平级的三个独立判断。现在逻辑对，是因为前面有 continue 挡住了重叠的数——一旦哪天删掉 continue，这三行就全错了。靠上下文才成立的写法，是最危险的一类。

7-17 · 双十一打折难点：骨架怎么搭

规则（反推）

输入 n 组数据。每组先给件数 k，再给 k 个价格。这一组价格求和后按满减结算，输出实付金额。

总价 S减免实付
0 – 990S
100 – 19930S − 30
200 – 29970S − 70
300 – 399110S − 110
400 及以上160S − 160

你的写法

#include<stdio.h>
int main(void)
{
    int n=0;
    int S=0;
    int i=0;
    int j=0;
    int a=0;

    scanf("%d",&n);
    for(j=1;j<=n;j++){
        for(scanf("%d",&i);i!=0;i--){
            scanf("%d",&a);
            S=S+a;}
        switch(S/100)
        {
        case 0: S=S;
        break;
        case 1: S=S-30;
        break;
        case 2: S=S-70;
        break;
        case 3: S=S-110;
        break;
        default:S=S-160;
        }
        printf("%d\n",S);
        S=0;
    }
    return 0;
}

标准写法 · 2.0（推荐）

#include <stdio.h>

int main(void)
{
    int caseCount = 0;   /* 组数 */
    int goodsCount = 0;  /* 件数 */
    int price = 0;       /* 单价 */
    int sum = 0;         /* 总价 */
    int i = 0;
    int j = 0;

    if (scanf("%d", &caseCount) != 1) {
        return 1;
    }

    for (i = 0; i < caseCount; i++) {
        sum = 0;   /* 开头清零 */

        if (scanf("%d", &goodsCount) != 1) {
            return 1;
        }
        for (j = 0; j < goodsCount; j++) {
            if (scanf("%d", &price) != 1) {
                return 1;
            }
            sum += price;
        }

        /* 从高到低判，天然互斥 */
        if (sum >= 400) {
            sum -= 160;
        } else if (sum >= 300) {
            sum -= 110;
        } else if (sum >= 200) {
            sum -= 70;
        } else if (sum >= 100) {
            sum -= 30;
        }

        printf("%d\n", sum);
    }

    return 0;
}
标准写法 · 3.0（switch 派，有个隐藏前提） 
        /* 前提：档位等宽，每 100 一档 */
        switch (sum / 100) {
        case 0:
            break;   /* 不满 100 */
        case 1:
            sum -= 30;
            break;
        case 2:
            sum -= 70;
            break;
        case 3:
            sum -= 110;
            break;
        default:
            sum -= 160;
            break;
        }

它能把总价映射成档号，是因为档位恰好每 100 一档。题目改成"满 100 减 30、满 300 减 100"，sum/100 立刻散架，得推倒重写——而 2.0 只改四个数字。

差异清单

① 清零放在末尾，是"顺序耦合"

你的循环末尾 S=0;
标准循环开头 sum=0;

放末尾的意思是：下一组能不能干净开始，取决于这一组有没有顺利走到最后一行。哪天中间加个 continue 或提前 return，清零就跳过，下一组带着上一组的钱进账——不报错，只算错。放开头，每组自己负责自己的初值。

② 一行干三件事

你的for(scanf(...); i!=0; i--)
标准读件数 → for(j=0;j<goodsCount;j++)

读第一个数、判断终止、递减计数挤在一个位置，而 i 同时是"还剩几件"和"数到第几件"。读的人得先翻译一遍。拆开后每句只干一件事。

③ switch 里的空操作

你的case 0: S=S; break;
标准case 0: break;

S=S 什么也没做。为了占位而写的语句是纯噪音，会让读的人停下来想"这里是不是漏了什么"。

④ 变量名

你的S i j a
标准sum goodsCount price caseCount

这不是审美。变量名是你明天重读这段代码时唯一的线索——名字含糊，就只能靠推理去回忆，而推理会出错。

7-17 我做过对拍：你的 + 2.0 + 3.0 三份代码，601 组数据（总价 0–600 全区间）输出一字不差。所以这份标准解法不是在纠正你的错误，是换骨架。

三道题的共性这一节比前三节加起来都重要

把三道题并排放，有三件事重复出现。重复出现的东西不是手滑，是习惯。

共性一 · 输入检查，三道题一次都没做

题你的写法标准写法
7-15scanf("%d",&n);if (scanf(...) != 1)
7-16for(scanf(...); ...)while (scanf(...) == 1 && ...)
7-17for(scanf(...); ...)if (scanf(...) != 1)

三道题，三种写法，共同点是全都没有检查 scanf 的返回值。这不是某一题漏了，这是一个还没建立的习惯。

共性二 · 把 scanf 塞进 for 的初始化位置

7-16 和 7-17 都出现了这个写法。它语法上没错（只执行一次），但它是技巧，不是表达——它把"读入"这件属于流程的事，藏进了"循环初始化"这个属于计数的位置。一个习惯重复两次，就说明它是你的默认反应，以后还会继续出现。

共性三 · 变量名清一色单字母

7-15 的 i / j / a / S、7-16 的 a / A / B / C / cnt、7-17 的 S / i / j / a。有意思的是：你在标准答案里见到过 goodsCount、match、total 这些名字，但它们从来没出现在你自己的代码里。说明你"看得懂"但"没采用"。

共性四 · 只有 7-15 有复杂度问题

7-16、7-17 都是一遍扫描，效率没问题。7-15 卡在 O(n²) 是因为没看出递推——这是唯一一处"算法"层面的差距，其余全是"骨架"层面的。

把四件事收成一句话：你的失分不在"想不出解法"，而在"解法想出来之后的工程习惯"——输入会不会崩、初值谁来管、变量叫什么、这个循环能不能更省。
